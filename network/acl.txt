! ACL pada multilayer switch: blokir VLAN Security (40) ke VLAN Server (20)
ip access-list extended SECURITY-TO-SERVER
 deny ip 192.168.40.0 0.0.0.255 192.168.20.0 0.0.0.255
 permit ip any any
!
! Pasang di SVI VLAN 40 arah masuk
interface vlan 40
 ip access-group SECURITY-TO-SERVER in
!
! Verifikasi:
!   show access-lists
!   show ip interface vlan 40 | include access list
